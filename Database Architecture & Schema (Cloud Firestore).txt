institutes/ {instituteId}
  ├── name: string
  ├── code: string (Unique key e.g., "INST001")
  ├── status: "ACTIVE" | "SUSPENDED" | "PENDING"
  ├── createdAt: timestamp
  └── settings: map

users/ {uid}
  ├── email: string
  ├── role: "super_admin" | "institute_admin" | "student"
  ├── instituteId: string ("GLOBAL" for super_admin)
  ├── status: "ACTIVE" | "DISABLED"
  └── profile: { name: string, rollNumber?: string, class?: string }

exams/ {examId}
  ├── instituteId: string
  ├── title: string
  ├── status: "DRAFT" | "PUBLISHED" | "UNPUBLISHED"
  ├── startWindow: timestamp
  ├── endWindow: timestamp
  ├── durationMinutes: number
  ├── totalPoolQuestions: number
  ├── displayQuestionCount: number
  ├── positiveMarks: number
  ├── negativeMarks: number
  ├── passMarks: number
  ├── maxAttempts: number
  ├── allowedLanguages: array ["en", "hi"]
  └── assignedBatches: array ["ALL"]

questions/ {questionId}
  ├── instituteId: string
  ├── examId: string
  ├── questionText: map { "en": string, "hi"?: string }
  ├── options: array [ { id: "A", text: map }, { id: "B", text: map }, ... ]
  ├── correctAnswer: string ("A" | "B" | "C" | "D")  <-- Never exposed to client
  ├── explanation: map
  └── marks: number

examAttempts/ {attemptId}
  ├── attemptId: string
  ├── studentId: string
  ├── instituteId: string
  ├── examId: string
  ├── status: "IN_PROGRESS" | "SUBMITTED" | "AUTO_SUBMITTED" | "DISQUALIFIED" | "EXPIRED"
  ├── attemptStartTime: timestamp (Server timestamp)
  ├── attemptEndTime: timestamp (Server calculated threshold)
  ├── questionOrder: array of questionIds (Randomized once upon initialization)
  ├── optionOrders: map { questionId: ["C", "A", "D", "B"] }
  ├── answers: map { questionId: "A" }
  ├── score: number (Calculated server-side on submission)
  ├── percentage: number
  ├── totalViolations: number
  └── activeSessionId: string

securityEvents/ {eventId}
  ├── eventId: string
  ├── attemptId: string
  ├── examId: string
  ├── studentId: string
  ├── instituteId: string
  ├── eventType: "TAB_SWITCH" | "WINDOW_BLUR" | "COPY_ATTEMPT" | "PASTE_ATTEMPT" | "CONTEXT_MENU" | "DEVTOOLS_SHORTCUT" | "FULLSCREEN_EXIT" | "MULTIPLE_SESSION"
  ├── timestamp: timestamp (Server)
  └── metadata: map

auditLogs/ {logId}
  ├── actorId: string
  ├── actorRole: string
  ├── instituteId: string
  ├── action: string
  ├── targetId: string
  └── timestamp: timestamp
